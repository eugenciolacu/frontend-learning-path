Scope Chain & Lexical Scoping

Understand how JavaScript resolves variable lookups by traversing the scope chain outward.

The Scope Chain

inner() scope
city = "Lisbon"
↑ not found? look up ↑
outer() scope
country = "Portugal"
↑ not found? look up ↑
Global scope
planet = "Earth"
↑ not found? → ReferenceError

Interactive Scope Examples

1. Basic Scope Chain Traversal

const planet = "Earth";     // global

function outer() {
  const country = "Portugal"; // outer scope

  function inner() {
    const city = "Lisbon";   // inner scope
    // Can access ALL outer scopes:
    console.log(city);    // own scope
    console.log(country); // outer scope
    console.log(planet);  // global scope
  }

  inner();
  // console.log(city); // ❌ ReferenceError
}
outer();

2. Lexical Scoping — Scope Follows Code Position

const x = "global";

function outer() {
  const x = "outer";

  function inner() {
    // Scope chain: inner → outer → global
    // Finds 'x' in outer first
    console.log(x); // "outer"
  }

  return inner; // return the function
}

const fn = outer();
fn(); // still logs "outer", not "global"!
// Scope was set at DEFINITION time, not call time

3. Variable Shadowing

const name = "Global Alice";

function greet() {
  const name = "Local Bob"; // shadows global 'name'
  console.log(name); // "Local Bob" — inner shadows outer
}

greet();
console.log(name); // "Global Alice" — outer unchanged

4. Closure — Scope Chain Preserved After Return

function makeCounter(start = 0) {
  let count = start; // in makeCounter's scope

  return {
    increment() { count++; },
    decrement() { count--; },
    value()     { return count; }
  };
  // The returned object closes over 'count'
}

const counter = makeCounter(10);
counter.increment();
counter.increment();
counter.decrement();
console.log(counter.value()); // 11

5. Block Scope vs Function Scope

function scopeDemo() {
  var funcScoped = "function scope";
  let blockScoped = "outer block";

  {
    var funcScoped2 = "still function scope!";
    let blockScoped = "inner block shadows";
    const alsoBlock = "block";
    console.log(blockScoped); // "inner block shadows"
    console.log(alsoBlock);   // "block"
  }

  console.log(funcScoped2); // accessible — var!
  console.log(blockScoped); // "outer block" — unaffected
  // console.log(alsoBlock); // ❌ ReferenceError
}
scopeDemo();

6. IIFE — Immediately Invoked Function Expression

// IIFE creates a private scope.
// Variables inside don't pollute global scope.
const result = (function () {
  const secret = "private value";
  const count = 42;
  // 'secret' and 'count' are NOT accessible outside
  return { count }; // only expose what you want
})();

console.log(result.count); // 42
// console.log(result.secret); // undefined
// console.log(secret); // ❌ ReferenceError
